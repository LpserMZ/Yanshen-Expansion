
on_startup = {
	if = { #GRY初始化
		limit = {
			NOT = {
				has_global_flag = YE_GRY_startup_gflag
			}
		}
		Y09 = { country_event = { id = YE_GRY.1 } } #边方
		Y23 = { country_event = { id = YE_GRY.2 } } #朱樾
		4949 = { province_event = { id = YE_GRY.6  } }

		Y01 = {
			set_variable = {
				which = oni_special_operations_team_sortie_num
				value = 0
			}
		}

		if = {
			limit = {
				ai = no
				has_estate = estate_eunuchs_anb
				NOT = {
					has_country_flag = anti_eunuch_country
				}
			}
			set_country_flag = anti_eunuch_country
		}
		YC1 = { remove_country = yes }

		Y18 = { country_event = { id = YE_GRY.57 days = 10 } }

		YE_add_special_prov_modifier = yes
		
		set_global_flag = YE_GRY_startup_gflag
	}
}

on_siege_lost_province =  {
	if = { # 瓜分古朔关
		limit = {
			is_capital_of = Y17
			owner = {
				is_in_war = {
					attackers = FROM
					casus_belli = cb_imperial
				}
			}
		}

		if = {
			limit = {
				owner = {
					is_in_war = {
						attacker_leader = Y15
						casus_belli = cb_imperial
					}
				}
			}
			Y15 = { country_event = { id = YE_GRY.52 } }
		}
		else_if = {
			limit = {
				owner = {
					is_in_war = {
						attacker_leader = Y05
						casus_belli = cb_imperial
					}
				}
			}
			Y05 = { country_event = { id = YE_GRY.52 } }
		}

	}
	if = { # 统一湮州
		limit = {
			OR = {
				superregion = yanshen_superregion
				province_id = 8411
				province_id = 8412
			}
			NOT = { area = ml_cave5_area }
		}

		if = {
			limit = {
				controller = {
					OR = { #与cb_yanzhou_unification条件一致
						is_yan_rank_II_tag = yes
						has_country_modifier = awareness_of_yan_nation 
					}

					OR = {
						is_in_war = {
							attacker_leader = this
							casus_belli = cb_yanzhou_unification
						}
						AND = {
							is_in_war = {
								defender_leader = this
								casus_belli = cb_yanzhou_unification
							}
							is_core = ROOT
						}
					}
					NOT = { owns = ROOT } #NOT = { is_core = ROOT }	
				}
			}

			cede_province = FROM
			controller = {
				country_event = {
					id = YE_GRY.54 
				}
			}
		}

	}
	if = { # 士兵热情-围城
		limit = {
			FROM = { has_government_mechanic = YE_blue_dream_soilder_fervor }
			fort_level = 1
		}
		if = { limit = { fort_level = 8 } FROM = { add_government_power = { mechanic_type = YE_blue_dream_soilder_fervor power_type = soilder_fervor value = 4.00 } } }
		else_if = { limit = { fort_level = 7 } FROM = { add_government_power = { mechanic_type = YE_blue_dream_soilder_fervor power_type = soilder_fervor value = 3.50 } } }
		else_if = { limit = { fort_level = 6 } FROM = { add_government_power = { mechanic_type = YE_blue_dream_soilder_fervor power_type = soilder_fervor value = 3.00 } } }
		else_if = { limit = { fort_level = 5 } FROM = { add_government_power = { mechanic_type = YE_blue_dream_soilder_fervor power_type = soilder_fervor value = 2.50 } } }
		else_if = { limit = { fort_level = 4 } FROM = { add_government_power = { mechanic_type = YE_blue_dream_soilder_fervor power_type = soilder_fervor value = 2.00 } } }
		else_if = { limit = { fort_level = 3 } FROM = { add_government_power = { mechanic_type = YE_blue_dream_soilder_fervor power_type = soilder_fervor value = 1.50 } } }
		else_if = { limit = { fort_level = 2 } FROM = { add_government_power = { mechanic_type = YE_blue_dream_soilder_fervor power_type = soilder_fervor value = 1.00 } } }
		else_if = { limit = { fort_level = 1 } FROM = { add_government_power = { mechanic_type = YE_blue_dream_soilder_fervor power_type = soilder_fervor value = 0.50 } } }		
	}
}

on_yearly_pulse = {
	if = { #yannification
		limit = {
			has_country_flag = allow_yannification_flag
		}
		every_owned_province = { 
			limit = {
				has_province_flag = prov_yannification_flag # 无法检测has_province_modifier = yannification_modifier 
				NOT = { culture_group = yan }
			}
			change_variable = { which = prov_yannification value = 1.5 }
			if = {
				limit = {
					check_variable = { which = prov_yannification_intermarriage_years value = 0.1 }
				}
				change_variable = { which = prov_yannification value = 3 }
				change_variable = { which = prov_yannification_intermarriage_years value = -1 }
			}
			if = {
				limit = {
					check_variable = { which = prov_yannification value = 100 }
				}
				clear_yannification_progress = yes

				yanification_change_culture = yes
			}
		}
	}
}

on_province_culture_changed = {
    if = {
        limit = {
            culture_group = yan

            OR = {
                has_province_flag = prov_yannification_flag

                has_province_modifier = yannification_unrest_modifier
                has_province_modifier = yannification_development_modifier
                has_province_modifier = yannification_intermarriage_modifier

                check_variable = {
                    which = prov_yannification
                    value = 0.1
                }
				check_variable = {
                    which = prov_yannification_intermarriage_years
                    value = 0.1
                }
            }
        }

        clear_yannification_progress = yes
    }
}
