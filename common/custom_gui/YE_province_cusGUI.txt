				NOT = {

				always = yes

		}

			#直属州中的任意省份可以
			if = {
				limit = {
					owner = {
						NOT = { has_country_flag = allow_prov_button_SLOT5_migrate_pop_2_flag }
					}
				}
				custom_trigger_tooltip = {
					tooltip = prov_button_SLOT5_migrate_pop_2_unfilled_tooltip
					always = no
				}
			}
			else_if = {
				limit = {
					any_province_in_state = {
						development = 18
						NOT = { has_province_modifier = rutual_xia_engineering_team }

					}
				}
				hidden_trigger = {
					always = yes
				}
			}

				custom_trigger_tooltip = {
					tooltip = state_condition_unfilled_tooltip
					always = no
				}
			}
			 
		}
	}

		if = { #小于18
			limit = {
				NOT = {
					development = 18
				}
			}

				add_base_tax = -1
				add_base_production = -1
				add_base_manpower = -1
			}

			add_base_production = 1
			add_base_manpower = 1
		}

			custom_tooltip = migrate_pop_effect_18_plus_tooltip
			area = {
				limit = {
					development = 18
					NOT = { has_province_modifier = rutual_xia_engineering_team }

					name = rutual_xia_engineering_team
					duration = 1825
				}
			}

				set_variable = {
					which = prov_yannification_intermarriage_years

				}
