add_yannification_progress_in_area_script_effect = {
    area = {
        #type = all
        limit = {
            has_province_flag = prov_yannification_flag
            NOT = {
                culture_group = yan
            }
        }

        change_variable = {
            which = prov_yannification
            value = $value$
        }
		check_yannification_completion_script_effect = yes
    }
}
clear_yannification_progress = {
    set_variable = {
        which = prov_yannification
        value = 0
    }

    set_variable = {
        which = prov_yannification_intermarriage_years
        value = 0
    }

    remove_province_modifier = yannification_unrest_modifier
    remove_province_modifier = yannification_development_modifier
    remove_province_modifier = yannification_intermarriage_modifier

    clr_province_flag = prov_yannification_flag
}
start_yannification_in_area_script_effect = {
    area = {
        #type = all
        limit = {
            NOT = {
                culture_group = yan
            }
			NOT = {
				has_province_flag = prov_yannification_flag
			}
        }

        custom_tooltip = start_yannification_in_area_script_effect_tooltip
		hidden_effect = {
			set_variable = {
				which = prov_yannification_intermarriage_years
				value = 0
			}
			set_variable = {
				which = prov_yannification
				value = 0
			}
			set_province_flag = prov_yannification_flag

			add_province_modifier = {
				name = yannification_development_modifier
				duration = -1
			}
			
		}    
		add_province_modifier = {
			name = yannification_unrest_modifier
			duration = -1
		}
    }
}
remove_yannification_unrest_in_area_script_effect = {
    area = {
        #type = all
        limit = {
            has_province_flag = prov_yannification_flag
        }

        remove_province_modifier = yannification_unrest_modifier
    }
}
check_yannification_completion_script_effect = {
    if = {
        limit = {
            has_province_flag = prov_yannification_flag

            # 依照你原本 yearly pulse 的写法：
            # 达到 100 或更高时成立。
            check_variable = {
                which = prov_yannification
                value = 100
            }
        }

        clear_yannification_progress = yes

        YE_yanification_change_culture = yes
    }
}
