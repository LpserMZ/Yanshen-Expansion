	get_core_in_yanzhou = {
		major = yes
		potential = {
			is_yan_rank_II_tag = yes
			NOT = { tag = Y98 }
			NOT = { has_country_flag = already_get_core_in_yanzhou_flag }
		}
		allow = {
			num_of_owned_provinces_with = {
				value = 160
				superregion = yanshen_superregion
			}
		}
		effect = {
			custom_tooltip = get_core_in_yanzhou_tooltip
			hidden_effect = {
				every_province = {
					limit = {
						OR = {
							superregion = yanshen_superregion
							province_id = 8411
							province_id = 8412
						}
						NOT = { area = ml_cave5_area }
						NOT = { owned_by = ROOT }
					}
					add_territorial_core = ROOT
				}
			}
			every_country = {
				limit = {
					capital_scope = { superregion = yanshen_superregion }
					NOT = {
						is_subject_of = ROOT
					}
					NOT = {
						tag = ROOT
					}
				}
				country_event = {
					id = YE_GRY.61
				}
			}
			every_subject_country = {
				add_liberty_desire = -100
			}
			set_country_flag = already_get_core_in_yanzhou_flag
		}
		ai_will_do = {
			factor = 1
		}
	}
